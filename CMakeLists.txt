cmake_minimum_required(VERSION 3.16)
project(Chunkpress VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(THREADS_PREFER_PTHREAD_FLAG ON)

option(CHUNKPRESS_GUI "Build the Qt Widgets desktop application" ON)
find_package(Threads REQUIRED)
find_package(ZLIB 1.2.9 REQUIRED)

function(chunkpress_warnings target)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endfunction()

add_library(chunkpress_core src/archive.cpp src/pipeline.cpp)
target_include_directories(chunkpress_core PUBLIC include)
target_compile_definitions(chunkpress_core PRIVATE _FILE_OFFSET_BITS=64)
target_link_libraries(chunkpress_core PUBLIC Threads::Threads ZLIB::ZLIB)
chunkpress_warnings(chunkpress_core)

add_executable(chunkpress-cli src/cli.cpp)
target_link_libraries(chunkpress-cli PRIVATE chunkpress_core)
chunkpress_warnings(chunkpress-cli)

if(CHUNKPRESS_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    add_executable(chunkpress src/gui.cpp assets/icons.qrc)
    set_target_properties(chunkpress PROPERTIES AUTORCC ON)
    target_link_libraries(chunkpress PRIVATE chunkpress_core Qt${QT_VERSION_MAJOR}::Widgets)
    chunkpress_warnings(chunkpress)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(chunkpress-tests tests/core.cpp)
    target_link_libraries(chunkpress-tests PRIVATE chunkpress_core)
    chunkpress_warnings(chunkpress-tests)
    add_test(NAME core COMMAND chunkpress-tests)
    set_tests_properties(core PROPERTIES TIMEOUT 60)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py" "$<TARGET_FILE:chunkpress-cli>")
        set_tests_properties(cli PROPERTIES TIMEOUT 45)
    endif()
    if(CHUNKPRESS_GUI)
        add_test(NAME gui COMMAND chunkpress --smoke-test)
        set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
    endif()
endif()
